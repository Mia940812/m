e.parameter.action : ''; var result = {}; if (action === 'getAllData' || !action) { result = { status: 'success', dailyReports: getSheetData('每日報數'), accounts: getSheetData('人員名冊'), shifts: getSheetData('14天報更').length > 0 ? getSheetData('現場違規') : getSheetData('異常工單') }; } else if (action === 'getDailyReports') { result = getSheetData('每日報數'); } else if (action === 'getAccounts') { result = getSheetData('人員名冊'); } else if (action === 'getShifts') { result = getSheetData('14天報更'); } else if (action === 'getRoster') { result = getSheetData('排更表格'); } else if (action === 'getCounts') { result = getSheetData('各區貨量'); } else if (action === 'getNewRecruits') { result = getSheetData('預約新人'); } else { result = { status: 'success', message: '後端運作正常', timestamp: new Date() }; } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var postData = {}; try { postData = JSON.parse(e.postData.contents); } catch (err) { postData = (e && e.parameter) ? data : []); var supRows = []; supList.forEach(function(s) { supRows.push([s.date || '', s.buyer || '', s.item || '', Number(s.qty || 1), Number(s.amount || 0), nowStr]); }); if (supRows.length > 0) sheetSup.getRange(2, 1, supRows.length, 6).setValues(supRows); } // 6. 件數流向 if (moduleName.includes('流向') || data.packageFlow) { var sheetF = ss.getSheetByName('件數流向') || ss.insertSheet('件數流向'); sheetF.clear(); sheetF.appendRow(['分區', '日期', '運單號碼', '圍欄編號', '負責步兵', '流向狀態', '備註說明', '更新時間']); var fList = data.packageFlow || (Array.isArray(data) ? data : []); var vRows = []; vList.forEach(function(v) { vRows.push([v.date || '', v.staff || '', v.reason || v['違規事項'] || '', Number(v.penalty || 0), v.status || '已登記', v.note || '', nowStr]); }); if (vRows.length > 0) sheetV.getRange(2, 1, vRows.length, 7).setValues(vRows); } } return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON); }
