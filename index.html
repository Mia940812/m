// 📈 營運利潤修復版（自動防呆計算，即點即開） function openModalProfit() { const eastC = Number(document.getElementById('repEastCount')?.value) || 0; const ktLgC = Number(document.getElementById('repKtLgCount')?.value) || 0; const ktIndC = Number(document.getElementById('repKtIndCount')?.value) || 0; const stC = Number(document.getElementById('repStCount')?.value) || 0; const totalPieces = eastC + ktLgC + ktIndC + stC; // 若當天尚未輸入報數，預設以基準估算（避免空白出 0） const rev = totalPieces > 0 ? ((eastC * 11.5) + (ktLgC * 11.0) + (ktIndC * 10.5) + (stC * 10.5)) : 3850.00; const cost = totalPieces > 0 ? ((eastC * 8.5) + (ktLgC * 8.0) + (ktIndC * 7.5) + (stC * 7.5)) : 2680.00; const net = rev - cost; const revEl = document.getElementById('profit-view-rev'); const costEl = document.getElementById('profit-view-cost'); const netEl = document.getElementById('profit-view-net'); if (revEl) revEl.textContent = `HK$ ${rev.toFixed(2)} ${totalPieces > 0 ? `(${totalPieces}票)` : '(預估)'}`; if (costEl) costEl.textContent = `HK$ ${cost.toFixed(2)}`; if (netEl) netEl.textContent = `HK$ ${net.toFixed(2)}`; openModal('modal-profit-calc'); }
