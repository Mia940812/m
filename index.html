function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var key = "派件報數"; var dataStr = ""; if (e && e.parameter && e.parameter.data) { dataStr = e.parameter.data; key = e.parameter.key || key; } else if (e && e.postData && e.postData.contents) { try { var p = JSON.parse(e.postData.contents); key = p.key || key; dataStr = typeof p.data === 'string' ? p.data : JSON.stringify(p.data); } catch(err) { dataStr = e.postData.contents; } } var now = new Date(); var parsed = null; try { parsed = JSON.parse(dataStr); } catch(err) {} // 1. 每日營運利潤核算表 (獨立分頁) if (key === "每日利潤核算" && parsed && parsed.date) { var sheet = getOrCreateSheet(ss, "每日營運利潤表", [ "核算日期", "當天總收入", "步兵出糧總額", "當日物資支出", "當日Claim報銷", "步兵扣數還款", "實質純利潤", "記錄時間" ]); sheet.appendRow([ parsed.date, parsed.grossRevenue, parsed.salaryPayout, parsed.suppliesExp, parsed.claimExp, parsed.advanceRepay, parsed.netProfit, now ]); return ContentService.createTextOutput("OK"); } // 2. 詳細分區埋數表 (獨立分頁) if (key === "詳細分區埋數" && parsed && parsed.date) { var sheet = getOrCreateSheet(ss, "詳細分區埋數表", [ "埋數日期", "全組總票數", "我方總應收", "東區票數", "東區單價", "東區步兵分工", "觀塘工商票數", "觀塘工商單價", "觀塘工商步兵分工", "觀塘麗港城票數", "觀塘麗港城單價", "觀塘麗港城步兵分工", "沙田F58票數", "沙田F58單價", "沙田F58步兵分工", "總Claim報銷", "雜費支出", "埋數備註", "記錄時間" ]); sheet.appendRow([ parsed.date, parsed.totalPieces, parsed.totalRevenue, parsed.eastPieces, parsed.eastUnitPrice, parsed.eastRiders, parsed.ktIndPieces, parsed.ktIndUnitPrice, parsed.ktIndRiders, parsed.ktLgPieces, parsed.ktLgUnitPrice, parsed.ktLgRiders, parsed.stPieces, parsed.stUnitPrice, parsed.stRiders, parsed.claimTotal, parsed.miscExp, parsed.note, now ]); return ContentService.createTextOutput("OK"); } // 3. 步兵報數與薪酬結算 if (key === "派件報數" && parsed && parsed.date) { var sheet = getOrCreateSheet(ss, "薪酬結算與報數總表", [ "派送日期", "人員 / 身份", "東區票數", "麗港城票數", "觀塘工商票數", "總票數", "車津", "Claim報銷", "Bonus獎金", "罰款", "扣數還款", "實發總工資", "記錄時間" ]); sheet.appendRow([ parsed.date, parsed.staff, parsed.eastCount || 0, parsed.ktLgCount || 0, parsed.ktIndCount || 0, parsed.totalCount || 0, parsed.carAllowance || 0, parsed.claim || 0, parsed.bonus || 0, parsed.penalty || 0, parsed.deductAmount || 0, parsed.totalSalary || 0, now ]); return ContentService.createTextOutput("OK"); } var defaultSheet = getOrCreateSheet(ss, "系統總紀錄", ["功能類型", "內容數據", "記錄時間"]); defaultSheet.appendRow([key, dataStr, now]); return ContentService.createTextOutput("OK"); } function getOrCreateSheet(ss, sheetName, headers) { var sheet = ss.getSheetByName(sheetName); if (!sheet) sheet = ss.insertSheet(sheetName); if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.getRange(1, 1, 1, headers.length).setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold"); } return sheet; } function doGet(e) { return doPost(e); }
