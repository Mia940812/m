function applyRolePermissions() { const isLogged = !!currentUser; const isMia = currentUser && (currentUser.name === 'Mia' || currentUser.isMia === true || (currentUser.role && currentUser.role.includes('總管理員'))); // 1. 步兵與大眾公開模組（登入後即可正常查看） const publicWorkModules = [ 'tab-progress', // 各區工作進度 'tab-assignments', // 區域分工排班 'tab-drivers', // 司機車線 'tab-supplies', // 物資採購 'tab-leaderboard', // 罰款獎池與龍虎榜 'tab-tickets', // 異常工單中心 'tab-packageflow', // 每日清場/滯留/退件流向 'tab-roster' // 14天公仔報更 ]; // 2. Mia 專屬隱私模組（利潤分析、出糧核對卡） // 透過導航列按鈕的文字與 ID 做精準隱藏 document.querySelectorAll('button, a, div.tab-pane').forEach(el => { const txt = el.innerText || ''; const id = el.id || ''; // 如果包含 Mia專屬 / 利潤 / 出糧卡 if (txt.includes('利潤分析') || txt.includes('Mia專屬') || txt.includes('出糧核對') || id.includes('salary-verify') || id.includes('tab-profit')) { el.style.display = isMia ? '' : 'none'; // 只有 Mia 先睇到，步兵直接隱藏 } }); // 3. 步兵只讀，Mia 可編輯（例如分工下拉、工資修改按鈕） document.querySelectorAll('.admin-edit-action').forEach(el => { el.style.display = isMia ? '' : 'none'; }); // 4. 更新頂部登入提示字樣 const badge = document.getElementById('login-status-badge') || document.querySelector('.login-badge'); if (badge) { if (isMia) { badge.innerHTML = `👑 Mia (總管理員模式)`; } else if (isLogged) { badge.innerHTML = `👤 隊員: ${currentUser.name} (已登入)`; } else { badge.innerHTML = `🔒 訪客模式 (未登入)`; } } }
